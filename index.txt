    <script>
        // Initialization
        const canvas = new fabric.Canvas('board', {
            isDrawingMode: true,
            selection: true,
            preserveObjectStacking: true
        });

        function resizeCanvas() {
            canvas.setWidth(window.innerWidth);
            canvas.setHeight(window.innerHeight);
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        // State variables
        let currentMode = 'pen';
        let currentTheme = 'dark';
        let isDashed = false;
        let isNeon = false;
        let showGrid = false;
        let gridGroup = null;
        let strokeColor = '#ffffff';
        let strokeWidth = 3;
        let isPanning = false;
        let lastPanPos = { x: 0, y: 0 };
        let shiftPressed = false;
        let startPoint = null;
        let tempLine = null;

        // Smooth Drawing Brush (Pencil with decimation)
        const penBrush = new fabric.PencilBrush(canvas);
        penBrush.decimation = 2; // Smooths out tablet fluctuations
        canvas.freeDrawingBrush = penBrush;

        // Theme and Grid logic
        function updateTheme() {
            const root = document.documentElement;
            root.setAttribute('data-theme', currentTheme);
            const bgColor = currentTheme === 'dark' ? '#121212' : '#f3f3f3';
            canvas.backgroundColor = bgColor;
            
            // Invert colors of existing objects
            const objects = canvas.getObjects();
            objects.forEach(obj => {
                if (obj.isGrid) return;
                const invert = (color) => {
                    if (!color) return color;
                    if (color === '#ffffff' && currentTheme === 'light') return '#000000';
                    if (color === '#000000' && currentTheme === 'dark') return '#ffffff';
                    return color;
                };
                if (obj.stroke) obj.set('stroke', invert(obj.stroke));
                if (obj.fill && obj.type !== 'path') obj.set('fill', invert(obj.fill));
            });
            
            strokeColor = currentTheme === 'dark' ? '#ffffff' : '#000000';
            document.getElementById('color-picker').value = strokeColor;
            updateBrush();
            if (showGrid) drawGrid(); else canvas.renderAll();
        }

        function drawGrid() {
            if (gridGroup) canvas.remove(gridGroup);
            if (!showGrid) { canvas.renderAll(); return; }
            
            const gridLines = [];
            const gridColor = currentTheme === 'dark' ? '#333333' : '#cccccc';
            const size = 40;
            
            for (let i = 0; i < (canvas.width / canvas.getZoom()); i += size) {
                gridLines.push(new fabric.Line([i, -canvas.height*2, i, canvas.height*2], { stroke: gridColor, selectable: false, isGrid: true }));
            }
            for (let i = 0; i < (canvas.height / canvas.getZoom()); i += size) {
                gridLines.push(new fabric.Line([-canvas.width*2, i, canvas.width*2, i], { stroke: gridColor, selectable: false, isGrid: true }));
            }
            
            gridGroup = new fabric.Group(gridLines, { selectable: false, evented: false, isGrid: true });
            canvas.add(gridGroup);
            gridGroup.sendToBack();
            canvas.renderAll();
        }

        // Tools and Brush Update
        function updateBrush() {
            canvas.freeDrawingBrush.color = strokeColor;
            canvas.freeDrawingBrush.width = parseInt(strokeWidth);
            
            if (isDashed) {
                canvas.freeDrawingBrush.strokeDashArray = [strokeWidth * 2, strokeWidth * 2];
            } else {
                canvas.freeDrawingBrush.strokeDashArray = null;
            }

            if (isNeon) {
                canvas.freeDrawingBrush.shadow = new fabric.Shadow({
                    blur: parseInt(strokeWidth) * 3,
                    offsetX: 0, offsetY: 0,
                    affectStroke: true,
                    color: strokeColor
                });
            } else {
                canvas.freeDrawingBrush.shadow = null;
            }
        }

        function setMode(mode) {
            currentMode = mode;
            document.querySelectorAll('.btn').forEach(b => b.classList.remove('active'));
            const btnMap = { 'pen':'btn-pen', 'eraser':'btn-eraser', 'select':'btn-select', 'pan':'btn-pan', 'laser':'btn-laser' };
            if(btnMap[mode]) document.getElementById(btnMap[mode]).classList.add('active');

            canvas.isDrawingMode = (mode === 'pen' || mode === 'laser');
            canvas.selection = (mode === 'select');
            
            if (mode === 'select') canvas.defaultCursor = 'default';
            else if (mode === 'pan') canvas.defaultCursor = 'grab';
            else if (mode === 'eraser') canvas.defaultCursor = 'crosshair';
            else canvas.defaultCursor = 'default';

            if (mode === 'laser') {
                canvas.freeDrawingBrush.color = '#ff0000';
                canvas.freeDrawingBrush.shadow = new fabric.Shadow({ blur: 15, color: '#ff0000' });
            } else if (mode === 'pen') {
                updateBrush();
            }
        }

        // Object Eraser & Laser Logic
        canvas.on('mouse:down', function(opt) {
            const evt = opt.e;
            if (currentMode === 'eraser' && opt.target) {
                canvas.remove(opt.target);
            }
            if (currentMode === 'pan' || (evt.buttons === 4) || (evt.buttons === 1 && evt.altKey)) {
                isPanning = true;
                lastPanPos = { x: evt.clientX, y: evt.clientY };
                canvas.defaultCursor = 'grabbing';
            }
            if (currentMode === 'pen' && shiftPressed) {
                canvas.isDrawingMode = false;
                let pointer = canvas.getPointer(opt.e);
                startPoint = pointer;
                tempLine = new fabric.Line([pointer.x, pointer.y, pointer.x, pointer.y], {
                    stroke: strokeColor,
                    strokeWidth: strokeWidth,
                    strokeDashArray: isDashed ? [strokeWidth*2, strokeWidth*2] : null
                });
                canvas.add(tempLine);
            }
        });

        canvas.on('mouse:move', function(opt) {
            if (isPanning) {
                const e = opt.e;
                const vpt = canvas.viewportTransform;
                vpt[4] += e.clientX - lastPanPos.x;
                vpt[5] += e.clientY - lastPanPos.y;
                canvas.requestRenderAll();
                lastPanPos = { x: e.clientX, y: e.clientY };
            }
            if (currentMode === 'pen' && shiftPressed && tempLine) {
                let pointer = canvas.getPointer(opt.e);
                tempLine.set({ x2: pointer.x, y2: pointer.y });
                canvas.renderAll();
            }
        });

        canvas.on('mouse:up', function(opt) {
            isPanning = false;
            if (currentMode === 'pan') canvas.defaultCursor = 'grab';
            
            if (currentMode === 'pen' && shiftPressed && tempLine) {
                canvas.isDrawingMode = true;
                tempLine = null;
            }
        });

        canvas.on('path:created', function(opt) {
            if (currentMode === 'laser') {
                const path = opt.path;
                setTimeout(() => {
                    fabric.util.animate({
                        startValue: 1, endValue: 0,
                        duration: 500,
                        onChange: (val) => { path.opacity = val; canvas.renderAll(); },
                        onComplete: () => canvas.remove(path)
                    });
                }, 1000);
            }
        });

        // Add Shapes & Text
        function addShape(type) {
            const center = canvas.getCenter();
            const opts = { left: center.left, top: center.top, fill: 'transparent', stroke: strokeColor, strokeWidth: parseInt(strokeWidth) };
            let obj;
            if (type === 'rect') obj = new fabric.Rect({ ...opts, width: 100, height: 100 });
            if (type === 'circle') obj = new fabric.Circle({ ...opts, radius: 50 });
            if (type === 'line') obj = new fabric.Line([50, 50, 200, 200], { stroke: strokeColor, strokeWidth: parseInt(strokeWidth) });
            canvas.add(obj);
            setMode('select');
        }

        document.getElementById('btn-rect').onclick = () => addShape('rect');
        document.getElementById('btn-circle').onclick = () => addShape('circle');
        document.getElementById('btn-line').onclick = () => addShape('line');

        document.getElementById('btn-text').onclick = () => {
            const text = new fabric.IText('Type here', { left: 100, top: 100, fill: strokeColor, fontFamily: 'Segoe UI' });
            canvas.add(text);
            canvas.setActiveObject(text);
            setMode('select');
        };

        document.getElementById('btn-sticky').onclick = () => {
            const group = new fabric.Group([
                new fabric.Rect({ width: 200, height: 200, fill: '#ffd700', rx: 10, ry: 10, shadow: new fabric.Shadow({blur: 5, color: 'rgba(0,0,0,0.3)', offsetX: 2, offsetY: 2}) }),
                new fabric.IText('Note', { left: 20, top: 20, fontSize: 20, fill: '#000' })
            ], { left: 100, top: 100 });
            canvas.add(group);
            setMode('select');
        };

        // DOM Event Listeners
        document.getElementById('btn-pen').onclick = () => setMode('pen');
        document.getElementById('btn-eraser').onclick = () => setMode('eraser');
        document.getElementById('btn-select').onclick = () => setMode('select');
        document.getElementById('btn-pan').onclick = () => setMode('pan');
        document.getElementById('btn-laser').onclick = () => setMode('laser');
        
        document.getElementById('color-picker').onchange = (e) => { strokeColor = e.target.value; updateBrush(); };
        document.getElementById('size-picker').oninput = (e) => { strokeWidth = e.target.value; updateBrush(); };
        
        document.getElementById('btn-dashed').onclick = (e) => { isDashed = !isDashed; e.target.classList.toggle('active'); updateBrush(); };
        document.getElementById('btn-neon').onclick = (e) => { isNeon = !isNeon; e.target.classList.toggle('active'); updateBrush(); };
        document.getElementById('btn-grid').onclick = (e) => { showGrid = !showGrid; e.target.classList.toggle('active'); drawGrid(); };
        
        document.getElementById('btn-theme').onclick = () => {
            currentTheme = currentTheme === 'dark' ? 'light' : 'dark';
            updateTheme();
        };

        // Save, Load, Export
        document.getElementById('btn-save').onclick = () => {
            const json = JSON.stringify(canvas.toJSON(['isGrid']));
            const blob = new Blob([json], {type: "application/json"});
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url; a.download = 'board.json'; a.click();
        };

        document.getElementById('btn-load').onclick = () => {
            document.getElementById('file-input').click();
        };

        document.getElementById('file-input').onchange = (e) => {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = (f) => {
                canvas.loadFromJSON(f.target.result, () => {
                    canvas.renderAll();
                    updateTheme(); // adjust loaded colors to current theme
                });
            };
            reader.readAsText(file);
        };

        document.getElementById('btn-export').onclick = () => {
            if (gridGroup) canvas.remove(gridGroup);
            const dataURL = canvas.toDataURL({ format: 'png', quality: 1, multiplier: 2 });
            const a = document.createElement('a');
            a.href = dataURL; a.download = 'board_export.png'; a.click();
            if (showGrid) drawGrid();
        };

        // Image Pasting
        window.addEventListener('paste', (e) => {
            if (e.clipboardData.items) {
                const items = e.clipboardData.items;
                for (let i = 0; i < items.length; i++) {
                    if (items[i].type.indexOf("image") !== -1) {
                        const blob = items[i].getAsFile();
                        const URLObj = window.URL || window.webkitURL;
                        const source = URLObj.createObjectURL(blob);
                        fabric.Image.fromURL(source, (img) => {
                            img.set({ left: 100, top: 100 });
                            img.scaleToWidth(400);
                            canvas.add(img);
                            setMode('select');
                        });
                    }
                }
            }
        });

        // Keyboard Shortcuts (Matching Tablet mappings)
        window.addEventListener('keydown', (e) => {
            if (e.key === 'Shift') shiftPressed = true;
            if (e.key.toLowerCase() === 'p') setMode('pen');
            if (e.key.toLowerCase() === 'e') setMode('eraser');
            if (e.key.toLowerCase() === 'v') setMode('select');
            if (e.key.toLowerCase() === 'h') setMode('pan');
            if (e.key.toLowerCase() === 't') document.getElementById('btn-text').click();
            if (e.key.toLowerCase() === 'n') document.getElementById('btn-sticky').click();
            if (e.key.toLowerCase() === 'l') setMode('laser');
            if (e.code === 'Space' && currentMode !== 'pan') {
                canvas.defaultCursor = 'grab';
                isPanning = true;
            }
            if ((e.key === 'Delete' || e.key === 'Backspace') && currentMode === 'select') {
                const activeObjects = canvas.getActiveObjects();
                if (activeObjects.length) {
                    activeObjects.forEach(obj => canvas.remove(obj));
                    canvas.discardActiveObject();
                }
            }
        });

        window.addEventListener('keyup', (e) => {
            if (e.key === 'Shift') shiftPressed = false;
            if (e.code === 'Space') {
                isPanning = false;
                setMode(currentMode); // Reset cursor
            }
        });

        // Mouse Wheel Zoom
        canvas.on('mouse:wheel', function(opt) {
            const delta = opt.e.deltaY;
            let zoom = canvas.getZoom();
            zoom *= 0.999 ** delta;
            if (zoom > 20) zoom = 20;
            if (zoom < 0.1) zoom = 0.1;
            canvas.zoomToPoint({ x: opt.e.offsetX, y: opt.e.offsetY }, zoom);
            opt.e.preventDefault();
            opt.e.stopPropagation();
        });

        // Setup Initial State
        updateTheme();
    </script>
